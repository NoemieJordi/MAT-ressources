\question[4]
Simplifie chaque fraction pour obtenir une fraction irréductible.

\vspace{10pt}
\setlength{\columnsep}{50pt}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=*,align=parleft, label=\alph*), itemsep=30pt]
\item $\dfrac{18}{30} =$ \sol{}{$\dfrac{3}{5}$}
\item $\dfrac{28}{42} =$ \sol{}{$\dfrac{2}{3}$}
\item $\dfrac{45}{60} =$ \sol{}{$\dfrac{3}{4}$}
\item $\dfrac{56}{64} =$ \sol{}{$\dfrac{7}{8}$}
\end{enumerate}
\end{multicols}
\setlength{\columnsep}{10pt}

\begin{solution}[0pt]
\textbf{1pt par réponse}\\[4pt]
0.5pt si la fraction est simplifiée mais pas jusqu'au bout.
\end{solution}
