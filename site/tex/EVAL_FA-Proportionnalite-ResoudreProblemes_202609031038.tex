\question[2]
Les prix pratiqués par ce cinéma sont-ils proportionnels au nombre de séances ?

\def\arraystretch{1.5}
\begin{center}\begin{tabularx}{0.5\textwidth}{l|C|C|C|}
Nombre de séances &1&4&14\\
\hline
Prix [CHF]&16,90&67,60&236,60\\
\end{tabularx}\end{center}
\def\arraystretch{1.5}

\vspace{10pt}{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
\begin{solutionorgrid}[140pt]

\vspace{5pt}Facteur de proportionnalité: 16,90 \hfill\textbf{(0.5pt)}

\vspace{5pt}$1 \cdot 16,90 = 16,90 \quad 4 \cdot 16,90 = 67,60 \quad 14 \cdot 16,90 = 236,60$\hfill\textbf{(1pt)}
\end{solutionorgrid}

\begin{tcolorbox}[enhanced jigsaw, interior hidden, colframe=box, parbox=false]%
Réponse: 
\sol{\vspace{20pt}}{Oui, les prix sont proportionnels au nombre de séances. \textbf{(0.5pt)}}
\end{tcolorbox}
