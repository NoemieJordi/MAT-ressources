\question[4]
Complète les égalités en amplifiant ou en simplifiant les fractions.

\vspace{10pt}
\setlength{\columnsep}{50pt}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=*,align=parleft, label=\alph*), itemsep=30pt]
\item $\dfrac{3}{4} = \dfrac{\sol{\hspace{20pt}}{15}}{20}$
\item $\dfrac{5}{6} = \dfrac{35}{\sol{\hspace{20pt}}{42}}$
\item $\dfrac{12}{18} = \dfrac{\sol{\hspace{20pt}}{2}}{3}$
\item $\dfrac{24}{36} = \dfrac{2}{\sol{\hspace{20pt}}{3}}$ 
\end{enumerate}
\end{multicols}
\setlength{\columnsep}{10pt}

\begin{solution}[0pt]
\textbf{1pt par réponse}\\[4pt]
a) amplifié par $5$ \quad b) amplifié par $7$ \quad c) simplifié par $6$ \quad d) simplifié par $12$ (0.5pt par terme)
\end{solution}
