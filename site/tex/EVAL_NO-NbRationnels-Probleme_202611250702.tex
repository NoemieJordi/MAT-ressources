\question[6]
Dans une classe de $24$ élèves, les $\dfrac{3}{8}$ des élèves font du football et $\dfrac{1}{4}$ des élèves font du basket. Les autres élèves ne pratiquent aucun de ces deux sports. Chaque élève pratique au plus un des deux sports.

\begin{parts}
	\vspace{20pt}\part
		Quelle fraction de la classe pratique le football ou le basket?
		
	\vspace{10pt}{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[80pt]
	
		\vspace{10pt}
		$\dfrac{3}{8} + \dfrac{1}{4} \textbf{\,(0.5pt)} = \dfrac{3}{8} + \dfrac{2}{8} \textbf{\,(0.5pt)} = \dfrac{5}{8} \textbf{\,(0.5pt)}$
	\end{solutionorgrid}
	
	\begin{tcolorbox}[enhanced jigsaw, interior hidden, colframe=box, parbox=false]%
	Réponse: 
	\sol{\vspace{20pt}}{$\dfrac{5}{8}$ de la classe pratique le football ou le basket. \textbf{\,(0.5pt)}}
	\end{tcolorbox}
		
	\vspace{20pt}\part
		Quelle fraction de la classe ne pratique aucun de ces deux sports?
		
	\vspace{10pt}{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[80pt]
	
		\vspace{10pt}
		$\dfrac{8}{8} - \dfrac{5}{8} \textbf{\,(0.5pt)} = \dfrac{3}{8}$
	\end{solutionorgrid}
	
	\begin{tcolorbox}[enhanced jigsaw, interior hidden, colframe=box, parbox=false]%
	Réponse: 
	\sol{\vspace{20pt}}{$\dfrac{3}{8}$ de la classe ne pratique aucun de ces deux sports. \textbf{\,(0.5pt)}}
	\end{tcolorbox}
		
	\vspace{20pt}\part
		Combien d'élèves ne pratiquent aucun de ces deux sports?
		
	\vspace{10pt}{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[80pt]
	
		\vspace{10pt}
		$24 : 8 = 3 \textbf{\,(1pt)}$
		
		\vspace{10pt}
		$3 \cdot 3 = 9 \textbf{\,(1pt)}$
	\end{solutionorgrid}
	
	\begin{tcolorbox}[enhanced jigsaw, interior hidden, colframe=box, parbox=false]%
	Réponse: 
	\sol{\vspace{20pt}}{$\mathbf{9}$ élèves ne pratiquent aucun de ces deux sports. \textbf{\,(1pt)}}
	\end{tcolorbox}
					
	\vfill
\end{parts}
