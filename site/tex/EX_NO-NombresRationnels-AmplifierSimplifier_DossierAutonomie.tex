\section{Vocabulaire}\label{sec:VocabulaireFractions}
 
\begin{regle}{Définition}
Une fraction s'écrit avec deux nombres séparés par un trait. Pour se comprendre, il y a trois mots de vocabulaire à connaître par c\oe ur : le \textcolor{themeColor}{numérateur}, le \textcolor{themeColor}{dénominateur} et la \textcolor{themeColor}{barre de fraction}.
\end{regle}
 
\vspace{15pt}
\begin{astuce}{Astuce}
La \textcolor{themeColor}{barre de fraction} est la barre entre les deux nombres : elle représente une division.\\[5pt]
Le \textcolor{themeColor}{numérateur} est le nombre du haut (celui qui est près des \textbf{nu}ages).\\[5pt]
Le \textcolor{themeColor}{dénominateur} est le nombre du bas, écrit en dessous de la barre de fraction.
\end{astuce}

\section{Lire une fraction}\label{sec:LireFraction}
 
\begin{regle}{Définition}
Pour \textcolor{themeColor}{lire une fraction}, on lit d'abord le numérateur, puis le dénominateur avec la terminaison \og ième \fg{}. On peut aussi lire \og $\ldots$ sur $\ldots$ \fg{}.
\[\dfrac{\textcolor{bleufraction}{2}}{\textcolor{vertfraction}{7}} \;:\; \textcolor{bleufraction}{\text{deux}}\;\textcolor{vertfraction}{\text{septièmes}} \qquad\text{ou}\qquad \textcolor{bleufraction}{\text{deux}}\text{ sur }\textcolor{vertfraction}{\text{sept}}\]
\end{regle}
 
\vspace{15pt}
\begin{astuce}{Attention}
Lorsque le dénominateur est $2$, $3$ ou $4$, on utilise les mots \textcolor{themeColor}{demi}, \textcolor{themeColor}{tiers} ou \textcolor{themeColor}{quart} (au pluriel s'il y en a plusieurs).
\end{astuce}

\providecommand{\GroupesCas}[1]{\begin{tikzpicture}
\foreach \g/\x/\y in {0/0/1.7,1/5/1.7,2/0/0,3/5/0}{
  \ifnum\g<#1 \fill[themeNO!20,rounded corners=8pt] (\x-0.3,\y-0.55) rectangle (\x+4.4,\y+0.55);\fi
  \draw[rounded corners=8pt,thick] (\x-0.3,\y-0.55) rectangle (\x+4.4,\y+0.55);
  \foreach \i in {0,1,2}{\node at (\x+0.6+\i*1.45,\y) {\casquette};}
}
\end{tikzpicture}}


\begin{regle}{Définition}
Une fraction permet de représenter une \textcolor{themeColor}{partie d'un ensemble}. Pour comprendre $\dfrac{\textcolor{bleufraction}{3}}{\textcolor{vertfraction}{4}}$ de $12$ casquettes, on procède en deux étapes.
\end{regle}

\vspace{15pt}
\begin{astuce}{Étape 1 : je regarde le dénominateur}
\[\dfrac{\textcolor{bleufraction}{3}}{\boxed{\textcolor{vertfraction}{4}}}\]
Le dénominateur indique en combien de groupes égaux je dois partager l'ensemble. Ici, je partage les $12$ casquettes en \textcolor{vertfraction}{$4$ groupes égaux}.

\vspace{5pt}
Comme $12 : 4 = 3$, chaque groupe contient $3$ casquettes.
\begin{center}
\GroupesCas{0}
\end{center}
\end{astuce}

\vspace{15pt}
\begin{astuce}{Étape 2 : je regarde le numérateur}
\[\dfrac{\boxed{\textcolor{bleufraction}{3}}}{\textcolor{vertfraction}{4}}\]
Le numérateur indique combien de groupes je dois prendre ou colorier. Ici, je colorie \textcolor{bleufraction}{$3$ groupes} sur les $4$.
\begin{center}
\GroupesCas{3}
\end{center}
\end{astuce}

\vspace{15pt}
\begin{regle}{À retenir}
Dans la fraction $\dfrac{\textcolor{bleufraction}{3}}{\textcolor{vertfraction}{4}}$, le \textcolor{vertfraction}{$4$} indique que je partage l'ensemble en $4$ groupes égaux, et le \textcolor{bleufraction}{$3$} indique que je prends $3$ de ces groupes.
\[\dfrac{3}{4}=\text{$3$ groupes pris sur $4$ groupes égaux}\]
\end{regle}

\section{Représenter une fraction sur une figure}\label{sec:RepresenterFigure}

\begin{regle}{Définition}
Pour représenter une fraction sur une figure, je partage la figure en \textcolor{themeColor}{parts égales}. Le dénominateur indique le nombre total de parts égales. Le numérateur indique le nombre de parts à colorier.
\end{regle}

\vspace{15pt}
\begin{astuce}{Stratégie}
Pour colorier $\dfrac{\textcolor{bleufraction}{3}}{\textcolor{vertfraction}{8}}$ de la figure :

\vspace{5pt}
\textcolor{themeColor}{1.} Je partage la figure en \textcolor{vertfraction}{$8$ parts égales}.\\
\textcolor{themeColor}{2.} Je colorie \textcolor{bleufraction}{$3$ parts}.

\begin{center}
\begin{tikzpicture}
\fill[themeNO!35] (0,0) rectangle (2.7,1.3);
\foreach \i in {1,...,7}{\draw[thick] (0.9*\i,0) -- (0.9*\i,1.3);}
\draw[very thick] (0,0) rectangle (7.2,1.3);
\draw[bleufraction,thick] (0,1.6) -- (2.7,1.6) (0,1.5) -- (0,1.7) (2.7,1.5) -- (2.7,1.7);
\node[bleufraction,above] at (1.35,1.65) {$3$ parts coloriées};
\draw[vertfraction,thick] (0,-0.3) -- (7.2,-0.3) (0,-0.2) -- (0,-0.4) (7.2,-0.2) -- (7.2,-0.4);
\node[vertfraction,below] at (3.6,-0.35) {$8$ parts égales};
\end{tikzpicture}
\end{center}

On a colorié $\dfrac{\textcolor{bleufraction}{3}}{\textcolor{vertfraction}{8}}$ de la figure.
\end{astuce}

\section{Comparer des fractions grâce au coloriage}\label{sec:ComparerColoriage}

\begin{regle}{Définition}
Pour \textcolor{themeColor}{comparer} deux fractions, on peut les représenter à l'aide d'un dessin. On dessine deux figures de \textcolor{themeColor}{même taille}, on partage chacune en autant de parts égales que l'indique le dénominateur, puis on colorie le nombre de parts indiqué par le numérateur.
\end{regle}

\vfill
\begin{astuce}{Stratégie}
\textcolor{themeColor}{1.} Je dessine deux figures de même taille.\\
\textcolor{themeColor}{2.} Je partage chaque figure en parts égales (dénominateur).\\
\textcolor{themeColor}{3.} Je colorie les parts correspondant aux fractions (numérateur).\\
\textcolor{themeColor}{4.} Je compare les deux parties coloriées.\\
\textcolor{themeColor}{5.} J'écris le bon signe : $<$, $>$ ou $=$.
\end{astuce}

\vfill
\begin{astuce}{Exemple}
Comparons $\dfrac{2}{3}$ et $\dfrac{3}{4}$.
\begin{center}
\begin{tabular}{r@{\hspace{0.6cm}}l}
$\dfrac{2}{3}$ & \GrilleFig{3}{1}{6}{1}{2}{blue!35} \\[18pt]
$\dfrac{3}{4}$ & \GrilleFig{4}{1}{6}{1}{3}{green!35}
\end{tabular}
\end{center}
La partie coloriée correspondant à $\dfrac{3}{4}$ est plus grande que celle correspondant à $\dfrac{2}{3}$. Donc :
\[\dfrac{2}{3}<\dfrac{3}{4}\]
\end{astuce}

\vfill
\begin{astuce}{Rappel : les signes de comparaison}
\def\arraystretch{2.4}
\begin{tabular}{@{}c@{\hspace{0.8cm}}l@{}}
$\dfrac{2}{3}<\dfrac{3}{4}$ & se lit \og $\dfrac{2}{3}$ est \textcolor{themeColor}{plus petit que} $\dfrac{3}{4}$ \fg{} \\
$\dfrac{3}{4}>\dfrac{2}{3}$ & se lit \og $\dfrac{3}{4}$ est \textcolor{themeColor}{plus grand que} $\dfrac{2}{3}$ \fg{} \\
$\dfrac{1}{2}=\dfrac{2}{4}$ & se lit \og $\dfrac{1}{2}$ est \textcolor{themeColor}{égal à} $\dfrac{2}{4}$ \fg{} \\
\end{tabular}

\vspace{5pt}
Le signe est comme une bouche ouverte : elle s'ouvre toujours du côté de la plus grande fraction.
\end{astuce}

\section{Comparer des fractions par le calcul}\label{sec:ComparerCalcul}

\begin{regle}{Définition}
Il existe une autre méthode que le coloriage : le \textcolor{themeColor}{calcul}. Pour comparer deux fractions, on les transforme pour qu'elles aient le même dénominateur (un \textcolor{themeColor}{dénominateur commun}). Il suffit alors de comparer les numérateurs.
\[\text{même dénominateur} \quad\Longrightarrow\quad \text{je compare les numérateurs}\]
\end{regle}

\vspace{15pt}
\begin{astuce}{Stratégie}
\textcolor{themeColor}{1.} Je cherche un dénominateur commun : le premier nombre qui se trouve dans le livret des deux dénominateurs.\\
\textcolor{themeColor}{2.} Je transforme les deux fractions en les amplifiant.\\
\textcolor{themeColor}{3.} Je compare les numérateurs et j'écris le bon signe.
\end{astuce}

\vspace{15pt}
\begin{astuce}{Exemple}
Comparons $\dfrac{\textcolor{bleufraction}{2}}{\textcolor{vertfraction}{3}}$ et $\dfrac{\textcolor{bleufraction}{3}}{\textcolor{vertfraction}{4}}$. Le premier nombre qui se trouve dans le livret de $3$ et dans le livret de $4$ est $12$. On transforme donc les deux fractions :
\begin{center}
\FlechesAmp{2}{3}{8}{12}{4} \qquad\qquad \FlechesAmp{3}{4}{9}{12}{3}
\end{center}
Les deux fractions ont maintenant le même dénominateur : il suffit de comparer les numérateurs. Comme $8<9$ :
\[\dfrac{2}{3}<\dfrac{3}{4}\]
\end{astuce}

\section{Amplifier une fraction}\label{sec:AmplifierFraction}

\begin{regle}{Définition}
\textcolor{themeColor}{Amplifier} une fraction, c'est multiplier son numérateur \textbf{et} son dénominateur par le même nombre (différent de 0). La fraction change d'écriture, mais pas de valeur.

\[\dfrac{2}{3}=\dfrac{2 \cdot 4}{3 \cdot 4}=\dfrac{8}{12}\]
\end{regle}

\vspace{20pt}
\begin{astuce}{Stratégie}
Pour trouver le facteur, on regarde les dénominateurs et on se demande :

\vspace{5pt}
{\centering\textit{\og $5$ fois combien font $20$ ?\fg{}} Ici $5 \cdot 4 = 20$, le facteur est $4$.\par}

\vspace{5pt}
On multiplie alors le numérateur par $4$ aussi.
\end{astuce}

\section{Simplifier une fraction}\label{sec:SimplifierFraction}

\[\dfrac{12}{18}=\dfrac{12 : 6}{18 : 6}=\dfrac{2}{3}\]
\end{regle}

Pour trouver le diviseur, on regarde les dénominateurs et on se demande :

\vspace{5pt}
On divise alors le numérateur par $5$ aussi.
\end{astuce}

\section{Chaînes de fractions égales}\label{sec:ChaineFractions}

\begin{regle}{Définition}
Une \textcolor{themeColor}{chaîne de fractions égales} est une suite de fractions qui ont toutes la même valeur. On passe d'une fraction à l'autre en amplifiant ou en simplifiant.
\[\dfrac{2}{3}=\dfrac{6}{9}=\dfrac{12}{18}=\dfrac{24}{36}\]
\end{regle}

\vspace{15pt}
\begin{astuce}{Stratégie}
Pour compléter une chaîne de fractions égales :

\vspace{5pt}
\textbf{1.} Repère la fraction complète (numérateur et dénominateur connus).\\
\textbf{2.} Compare chaque fraction à compléter avec la fraction complète (pas avec sa voisine) : \textit{\og $\ldots$ fois combien font $\ldots$ ?\fg{}}\\
\textbf{3.} Selon le cas, on amplifie ($\cdot$) ou on simplifie ($:$).
\end{astuce}

\section{Fractions irréductibles}\label{sec:FractionsIrreductibles}

\begin{regle}{Définition}
Une fraction est \textcolor{themeColor}{irréductible} quand on ne peut plus la simplifier : le seul diviseur commun de son numérateur et de son dénominateur est $1$.
\[\dfrac{6}{8}=\dfrac{6 : 2}{8 : 2}=\dfrac{3}{4}\]
$\dfrac{6}{8}$ n'est pas irréductible (on peut diviser par $2$), alors que $\dfrac{3}{4}$ est irréductible.
\end{regle}

\vspace{15pt}
\begin{astuce}{Stratégie}
Pour rendre une fraction irréductible :

\vspace{5pt}
\textbf{1.} Cherche un diviseur commun du numérateur et du dénominateur (autre que $1$).\\
\textbf{2.} Simplifie par ce diviseur.\\
\textbf{3.} Recommence jusqu'à ce qu'il n'y ait plus de diviseur commun.
\end{astuce}

\section{Additionner et soustraire : même dénominateur}\label{sec:AdditionnerMemeDenominateur}

\begin{regle}{Définition}
Pour \textcolor{themeColor}{additionner} ou \textcolor{themeColor}{soustraire} des fractions qui ont le \textbf{même dénominateur}, on additionne (ou on soustrait) les numérateurs et on \textbf{garde le dénominateur}.
\[\dfrac{3}{8}+\dfrac{2}{8}=\dfrac{3+2}{8}=\dfrac{5}{8} \qquad\qquad \dfrac{7}{9}-\dfrac{4}{9}=\dfrac{7-4}{9}=\dfrac{3}{9}=\dfrac{1}{3}\]
\end{regle}

\vspace{20pt}
\begin{astuce}{Stratégie}
Pense en parts :

\vspace{5pt}
{\centering\textit{$3$ huitièmes $+$ $2$ huitièmes $=$ $5$ huitièmes.}\par}

\vspace{5pt}
Le dénominateur dit de quelles parts on parle : il ne change pas. À la fin, simplifie le résultat si c'est possible.
\end{astuce}

\section{Additionner et soustraire : dénominateurs différents}\label{sec:AdditionnerDenominateurDifferent}

\begin{regle}{Définition}
On ne peut additionner ou soustraire que des fractions de même dénominateur. Sinon, on \textcolor{themeColor}{amplifie} les fractions pour obtenir un \textbf{dénominateur commun} : un multiple commun des deux dénominateurs.
\[\dfrac{1}{2}+\dfrac{1}{3}=\dfrac{1 \cdot 3}{2 \cdot 3}+\dfrac{1 \cdot 2}{3 \cdot 2}=\dfrac{3}{6}+\dfrac{2}{6}=\dfrac{5}{6}\]
\end{regle}

\vspace{20pt}
\begin{astuce}{Stratégie}
Pour calculer avec des dénominateurs différents :

\vspace{5pt}
\textbf{1.} Cherche un dénominateur commun (le plus petit possible) en listant les multiples des deux dénominateurs.\\
\textbf{2.} Amplifie chaque fraction pour obtenir ce dénominateur.\\
\textbf{3.} Additionne ou soustrais les numérateurs, et garde le dénominateur.\\
\textbf{4.} Simplifie le résultat si c'est possible.

\vspace{5pt}
\textit{Exemple :} pour $\dfrac{3}{4}-\dfrac{1}{6}$, les multiples de $4$ sont $4$, $8$, $12$ et ceux de $6$ sont $6$, $12$ : le dénominateur commun est $12$.
\end{astuce}

\vspace{20pt}

\hline
\cellcolor{themeNO!30}\textbf{Calcul} & \cellcolor{themeNO!30}\textbf{Dénominateur commun} & \cellcolor{themeNO!30}\textbf{Résultat} \\\hline
$\dfrac{1}{2}+\dfrac{1}{4}$ & $4$ & $\dfrac{3}{4}$ \\\hline
$\dfrac{1}{2}+\dfrac{1}{3}$ & $6$ & $\dfrac{5}{6}$ \\\hline
$\dfrac{3}{4}-\dfrac{1}{6}$ & $12$ & $\dfrac{7}{12}$ \\\hline

\section{Multiplier des fractions}\label{sec:MultiplierFractions}

\begin{regle}{Définition}
Pour \textcolor{themeColor}{multiplier} deux fractions, on multiplie les numérateurs entre eux et les dénominateurs entre eux. Pas besoin de dénominateur commun !
\[\dfrac{2}{3} \cdot \dfrac{4}{5}=\dfrac{2 \cdot 4}{3 \cdot 5}=\dfrac{8}{15}\]
\end{regle}

\vspace{20pt}
\begin{astuce}{Stratégie}
\textbf{Simplifie avant de calculer} : on peut simplifier un numérateur avec un dénominateur.
\[\dfrac{3}{4} \cdot \dfrac{8}{9}=\dfrac{3 \cdot 8}{4 \cdot 9}=\dfrac{1 \cdot 2}{1 \cdot 3}=\dfrac{2}{3}\]
\end{astuce}

\section{Diviser des fractions}\label{sec:DiviserFractions}

\begin{regle}{Définition}
L'\textcolor{themeColor}{inverse} d'une fraction s'obtient en échangeant son numérateur et son dénominateur : l'inverse de $\dfrac{a}{b}$ est $\dfrac{b}{a}$. Le nombre $0$ n'a pas d'inverse.\\[5pt]
\textcolor{themeColor}{Diviser} par une fraction, c'est multiplier par son inverse.
\[\dfrac{2}{3} : \dfrac{5}{7}=\dfrac{2}{3} \cdot \dfrac{7}{5}=\dfrac{14}{15}\]
\end{regle}

\vspace{20pt}
\begin{astuce}{Stratégie}
Un nombre entier s'écrit avec le dénominateur $1$ : $5=\dfrac{5}{1}$, donc l'inverse de $5$ est $\dfrac{1}{5}$.
\[\dfrac{3}{4} : 6=\dfrac{3}{4} \cdot \dfrac{1}{6}=\dfrac{3}{24}=\dfrac{1}{8} \qquad\qquad 6 : \dfrac{2}{3}=6 \cdot \dfrac{3}{2}=\dfrac{18}{2}=9\]
Ensuite, on calcule comme pour une multiplication : on simplifie avant de calculer si possible.
\end{astuce}
