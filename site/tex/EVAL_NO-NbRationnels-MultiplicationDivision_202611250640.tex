\question[10]
Effectue les calculs suivants. Montre tes étapes et donne le résultat sous la forme d'une fraction irréductible.

\vspace{10pt}

\begin{enumerate}[leftmargin=*,align=parleft, label=\alph*), itemsep=60pt]
\item $\dfrac{2}{5} \cdot \dfrac{3}{7} =$ \sol{}{$\dfrac{6}{35}$}
\item $\dfrac{5}{6} \cdot \dfrac{9}{10} =$ \sol{}{$\dfrac{3}{4}$}
\item $\dfrac{3}{4} \cdot \dfrac{2}{9} =$ \sol{}{$\dfrac{1}{6}$}
\item $6 \cdot \dfrac{5}{8} =$ \sol{}{$\dfrac{15}{4}$}
\item $\dfrac{3}{5} : \dfrac{2}{7} =$ \sol{}{$\dfrac{21}{10}$}
\item $\dfrac{8}{9} : \dfrac{4}{3} =$ \sol{}{$\dfrac{2}{3}$}
\end{enumerate}


\begin{solution}[0pt]
\textbf{a) et b) : 1pt par calcul.} \quad \textbf{c) à f) : 2pt par calcul} (1pt pour la méthode : multiplication posée, ou multiplication par l'inverse pour la division ; 1pt pour le résultat simplifié)\\[8pt]
a) $\dfrac{2 \cdot 3}{5 \cdot 7} = \dfrac{6}{35}$ \quad b) $\dfrac{5 \cdot 9}{6 \cdot 10} = \dfrac{45}{60} = \dfrac{3}{4}$\\[6pt]
c) $\dfrac{3 \cdot 2}{4 \cdot 9} = \dfrac{6}{36} = \dfrac{1}{6}$ \quad d) $\dfrac{6}{1} \cdot \dfrac{5}{8} = \dfrac{30}{8} = \dfrac{15}{4}$\\[6pt]
e) $\dfrac{3}{5} \cdot \dfrac{7}{2} = \dfrac{21}{10}$ \quad f) $\dfrac{8}{9} \cdot \dfrac{3}{4} = \dfrac{24}{36} = \dfrac{2}{3}$
\end{solution}
