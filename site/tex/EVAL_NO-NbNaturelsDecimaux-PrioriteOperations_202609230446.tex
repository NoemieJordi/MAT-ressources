\question
Effectue les calculs suivants \textbf{en notant toutes les étapes intermédiaires}.

\begin{parts}
\vspace{20pt}\part[2] $540 - 480:(110-30)\cdot 70=$

\vspace{10pt}{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[200pt]
		\begin{align*}
		540 - 480:(110-30)\cdot 70&=\\
		540 - 480:80\cdot 70&=&\textbf{0.5pt}\\
		540 - 6\cdot 70&=&\textbf{0.5pt}\\
		540 - 420&=&\textbf{0.5pt}\\
		120&&\textbf{0.5pt}
		\end{align*}
	\end{solutionorgrid}

\vspace{20pt}\part[3] $60 - 24:4\cdot (10 - 2\cdot4)^3=$

\vspace{10pt}{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[240pt]
		\begin{align*}
		60 - 24:4\cdot (10 - 2\cdot4)^3&=\\
		60 - 24:4\cdot (10 - 8)^3&=&\textbf{0.5pt}\\
		60 - 24:4\cdot 2^3&=&\textbf{0.5pt}\\
		60 - 24:4\cdot 8&=&\textbf{0.5pt}\\
		60 - 6\cdot 8&=&\textbf{0.5pt}\\
		60 - 48&=&\textbf{0.5pt}\\
		12&&\textbf{0.5pt}
		\end{align*}
	\end{solutionorgrid}
\end{parts}
