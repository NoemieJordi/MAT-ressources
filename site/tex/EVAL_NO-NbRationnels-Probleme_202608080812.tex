\question[3]
Un viticulteur (producteur de vin) vend les sept neuvièmes de sa production, à savoir 714 bouteilles de vin. 

\begin{parts}
	\vspace{20pt}\part
		Quelle fraction lui reste-t-il à vendre?
		
	\vspace{10pt}{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[120pt]
	
		\vspace{10pt}
		$\dfrac{9}{9} - \dfrac{7}{9} \textbf{\,(0.5pt)} = \dfrac{2}{9} \textbf{\,(0.5pt)}$
	\end{solutionorgrid}
	
	\begin{tcolorbox}[enhanced jigsaw, interior hidden, colframe=box, parbox=false]%
	Réponse: 
	\sol{\vspace{20pt}}{Il doit encore vendre \textbf{$\dfrac{2}{9}$} de sa production. \textbf{\,(0.5pt)}}
	\end{tcolorbox}
		
	\vspace{20pt}\part
		Combien de bouteilles a-t-il produit au total?
		
	\vspace{10pt}{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[120pt]
	
		\vspace{10pt}
		$714\div7 = 102 \textbf{\,(0.5pt)}$
		
		\vspace{10pt}
		$102\cdot9 = 918 \textbf{\,(0.5pt)}$
	\end{solutionorgrid}
	
	\begin{tcolorbox}[enhanced jigsaw, interior hidden, colframe=box, parbox=false]%
	Réponse: 
	\sol{\vspace{20pt}}{Il a produit \textbf{918} bouteille. \textbf{\,(0.5pt)}}
	\end{tcolorbox}		
					
	\vfill
\end{parts}
