\section*{Carte de suivi --- paliers}

\noindent \textbf{Nom~:} \dotfill

\bigskip
\begin{tcolorbox}[colback=\themeColor!5, colframe=\themeColor, title=Principe des paliers, fonttitle=\bfseries]
\begin{itemize}[leftmargin=1.5em, itemsep=3pt]
\item \textbf{Palier 1} (chaque semaine) --- palier de départ, ou retour après une erreur répétée.
\item \textbf{Palier 2} (une semaine sur deux) --- atteint après 2 réussites consécutives au palier 1.
\item \textbf{Palier 3} (une fois par mois, entretien) --- atteint après 2 réussites consécutives au palier 2.
\item Une erreur fait redescendre d'un palier ; une erreur répétée du même type ramène directement au palier 1 pour cette opération.
\end{itemize}
\end{tcolorbox}
