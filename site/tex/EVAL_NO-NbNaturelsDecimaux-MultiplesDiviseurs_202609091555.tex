Multiples et diviseurs

\begin{parts}
\vspace{10pt}\part[2] Écris les 6 premiers multiples de 17.

\vspace{10pt}{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[80pt]
	
		\vspace{10pt}
		M$_{17}$ = $\{0; 17; 34; 51; 68; 85; 102; 119; 136; 153; \ldots\}$
		
		\vspace{10pt}\textbf{1,5\,pt} -0.5\,pt par erreur et \textbf{0.5\,pt} si l'écriture est correcte.
	\end{solutionorgrid}
