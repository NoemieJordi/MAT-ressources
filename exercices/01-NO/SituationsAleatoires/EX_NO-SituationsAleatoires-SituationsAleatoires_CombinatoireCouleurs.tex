\header{Couleurs}

\begin{itemize}
\item[a)]Tu disposes d'un crayon rouge et d'un crayon bleu. De combien de façons différentes peux-tu colorier ce bonhomme en utilisant une ou deux de ces couleurs?
\end{itemize}

\sol{\vfill}{\vspace{30pt}}
\begin{center}\begin{tikzpicture}
%----tête----%
\tkzDefPoints{0/0.25/T1, 0.75/0.25/T2}
\tkzDrawCircle(T1,T2)
%----corps----%
\tkzDefPoints{-0.5/-0.5/C1, 0.5/-0.5/C2,0.5/-2/C3, -0.5/-2/C4}
\tkzDrawPolygon(C1,C2,C3,C4)
%----jambes----%
\tkzDefPoints{-1/-3.5/J1, 1/-3.5/J2}
\tkzDrawSegments(C4,J1 J1,J2 J2,C3)
\end{tikzpicture}\end{center}

\sol{\vfill
\begin{tikzpicture}
\draw[step=0.5cm, gray, very thin] (0,0) grid (\textwidth,17);
\end{tikzpicture}}{\color{black}\vspace{50pt}
