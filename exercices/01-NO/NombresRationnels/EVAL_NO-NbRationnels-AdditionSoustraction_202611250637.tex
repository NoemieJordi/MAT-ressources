\question[8]
Effectue les calculs suivants. Montre tes étapes et donne le résultat sous la forme d'une fraction simplifiée au maximum.

\vspace{10pt}
\begin{enumerate}[leftmargin=*,align=parleft, label=\alph*), itemsep=60pt]
\item $\dfrac{3}{4} + \dfrac{5}{6} =$ \sol{}{$\dfrac{19}{12}$}
\item $\dfrac{7}{10} - \dfrac{2}{5} =$ \sol{}{$\dfrac{3}{10}$}
\item $\dfrac{5}{6} - \dfrac{3}{8} =$ \sol{}{$\dfrac{11}{24}$}
\item $\dfrac{11}{15} - \dfrac{1}{3} =$ \sol{}{$\dfrac{2}{5}$}
\item $0.7 + \dfrac{3}{5} =$ \sol{}{$\dfrac{13}{10}$}
\item $0.75 - \dfrac{1}{6} =$ \sol{}{$\dfrac{7}{12}$}
\end{enumerate}

\begin{solution}[0pt]
\textbf{Partie A :} 1pt par calcul (0.5pt pour le même dénominateur, 0.5pt pour le résultat)\\[4pt]
\textbf{Partie B :} 2pt par calcul (0.5pt pour la fraction du nombre à virgule, 0.5pt pour le même dénominateur, 1pt pour le résultat)\\[8pt]
a) $\dfrac{9}{12} + \dfrac{10}{12} = \dfrac{19}{12}$ \quad b) $\dfrac{7}{10} - \dfrac{4}{10} = \dfrac{3}{10}$\\[6pt]
c) $\dfrac{20}{24} - \dfrac{9}{24} = \dfrac{11}{24}$ \quad d) $\dfrac{11}{15} - \dfrac{5}{15} = \dfrac{6}{15} = \dfrac{2}{5}$\\[6pt]
e) $\dfrac{7}{10} + \dfrac{6}{10} = \dfrac{13}{10}$ \quad f) $\dfrac{3}{4} - \dfrac{1}{6} = \dfrac{9}{12} - \dfrac{2}{12} = \dfrac{7}{12}$
\end{solution}
