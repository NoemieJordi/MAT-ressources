\question[4]
Effectue les calculs suivants. Donne le résultat sous la forme d'une fraction simplifiée au maximum.

\vspace{10pt}
\begin{enumerate}[leftmargin=*,align=parleft, label=\alph*), itemsep=40pt]
\item $\dfrac{3}{7} + \dfrac{2}{7} =$ \sol{}{$\dfrac{5}{7}$}
\item $\dfrac{11}{9} - \dfrac{4}{9} =$ \sol{}{$\dfrac{7}{9}$}
\item $\dfrac{1}{5} + \dfrac{3}{10} =$ \sol{}{$\dfrac{1}{2}$}
\item $\dfrac{7}{8} - \dfrac{1}{4} =$ \sol{}{$\dfrac{5}{8}$}
\end{enumerate}

\begin{solution}[0pt]
\textbf{1pt par calcul}\\[4pt]
a) $\dfrac{5}{7}$ \quad b) $\dfrac{7}{9}$\\[6pt]
c) $\dfrac{2}{10} + \dfrac{3}{10} = \dfrac{5}{10} = \dfrac{1}{2}$ (0.5pt pour le même dénominateur, 0.5pt pour le résultat simplifié)\\[6pt]
d) $\dfrac{7}{8} - \dfrac{2}{8} = \dfrac{5}{8}$ (0.5pt pour le même dénominateur, 0.5pt pour le résultat)
\end{solution}
