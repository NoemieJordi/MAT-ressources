\question[6]
Effectue les calculs suivants en notant les étapes intermédiaires.

\vspace{10pt}
\begin{parts}
\part $(-50) + (-50) - (+89) =$ 

	{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[70pt]
	$ -50 - 50 - 89 \hspace{5pt}\textbf{(0.5pt)}\hspace{5pt}= -189$\hspace{5pt}\textbf{(0.5 pt)}
	\end{solutionorgrid}
	
\vspace{10pt}
\part $(+143) - (-95) + (-26) =$ 

	{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[70pt]
	$+143 + 95 - 26 \hspace{5pt}\textbf{(0.5pt)} \hspace{5pt}= +238 - 26 \hspace{5pt}\textbf{(0.5pt)}\hspace{5pt}=+212$ \hspace{5pt}\textbf{(0.5pt)}
	\end{solutionorgrid}
	

\vspace{10pt}
\part $(-32) - (+164) + (+73) - (-37) =$ 

	{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[70pt]
$-32 - 164 + 73 + 37 \hspace{5pt}\textbf{(0.5pt)}\hspace{5pt}= +110 - 196 \hspace{5pt}\textbf{(0.5pt)}\hspace{5pt}= -86$ \hspace{5pt}\textbf{(0.5pt)}
	\end{solutionorgrid}
	
\vspace{10pt}
\part $(+8,4) + (-3,4) - (+1,5) - (+4,1) =$ 

	{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[70pt]
$+8,4 - 3,4 - 1,5 - 4,1  \hspace{5pt}\textbf{(0.5pt)}\hspace{5pt}= +12,5 - 4,9 \hspace{5pt}\textbf{(1pt)}\hspace{5pt}= -7,6$ \hspace{5pt}\textbf{(0.5pt)}
	\end{solutionorgrid}
\end{parts}
