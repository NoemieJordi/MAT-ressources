\question[7]
Calcule en indiquant tes étapes intermédiaires.

\begin{parts}

\vspace{20pt}\part $12^1 + 1^{12}=$

\vspace{10pt}{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[60pt]
	
		\vspace{10pt}
		$12 + 1 = 13$
		
		\textbf{1\,pt} -0,5\,pt par erreur
	\end{solutionorgrid}
	
\vspace{20pt}\part $2^7-2^5=$

\vspace{10pt}{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[60pt]
	
		\vspace{10pt}
		$128 - 32 = 96$
		
		\textbf{1\,pt} -0,5\,pt par erreur
	\end{solutionorgrid}
	
\vspace{20pt}\part $5^3 + 4^2 + 3^1=$

	\begin{solutionorgrid}[60pt]
	
		\vspace{10pt}
		$125 + 16 + 3 = 144$
		
		\textbf{1,5\,pt} -0,5\,pt par erreur
	\end{solutionorgrid}
