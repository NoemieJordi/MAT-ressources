Multiples et diviseurs

\begin{parts}

\vspace{20pt}{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[40pt]
	
		\vspace{10pt}
		M$(29)$ = $\{29; 58; 87; 116; 145; \ldots\}$
		
		\vspace{10pt}\textbf{1,5pt} -0.5pt par erreur et \textbf{0.5pt} si l'écriture est correcte.
	\end{solutionorgrid}
