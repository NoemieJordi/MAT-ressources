\end{tikzpicture}}

\end{align*}
}

{\bfseries\color{themeColor}Exercice 2.} \quad{\Large$\left[\left(6-8\right)^3 \div \left(-4\right)\right]^3 \cdot 2 + \left(-10\right) =$}

\sol{
\begin{tikzpicture}
\draw[step=0.5cm, gray, very thin] (0,0) grid (\textwidth,13);
\end{tikzpicture}}
%--------------solution--------------%
{\vspace{-30pt}\color{black}
\begin{align*}
\left[\left(6-8\right)^3 \div \left(-4\right)\right]^3 \cdot 2 + \left(-10\right) &=\\
\left[\left(\textcolor{themeColor}{-2}\right)^3 \div \left(-4\right)\right]^3 \cdot 2 + \left(-10\right) &=\\
\left[\left(\textcolor{themeColor}{-8}\right) \div \left(-4\right)\right]^3 \cdot 2 + \left(-10\right) &=\\
\textcolor{themeColor}{2}^3 \cdot 2 + \left(-10\right) &=\\
\textcolor{themeColor}{8} \cdot 2 + \left(-10\right) &=\\
\textcolor{themeColor}{16} + \left(-10\right) &=\\
\textcolor{themeColor}{6}
\end{align*}
}
