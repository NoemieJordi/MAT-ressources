\question
Effectue les calculs suivants \textbf{en notant toutes les étapes intermédiaires}.

\begin{parts}
	
\vspace{20pt}\part[1] $105 -75 : 3 \cdot 4=$

\vspace{10pt}{\footnotesize\raggedleft Espace pour ta démarche et tes calculs\par}
	\begin{solutionorgrid}[180pt]
		\begin{align*}
		105 -75 : 3 \cdot 4&=\\
		105 -25 \cdot 4&=\\
		105 -100&=\\
		5&
		\end{align*}
	
	\hfill\textbf{1\,pt} (-0,5\,pt par erreur)
	\end{solutionorgrid}

		13^2 + 2 \cdot (16-9) \cdot (3 + 2)^1 &=\\
		13^2 + 2 \cdot 7 \cdot 5^1 &=\\
		169 + 2 \cdot 7 \cdot 5 &=\\
		169 + 14 \cdot 5 &=\\
		169 + 70 &=\\
		239&
		\end{align*}
	
	\hfill\textbf{3\,pts} (-0,5\,pt par erreur)
	\end{solutionorgrid}
